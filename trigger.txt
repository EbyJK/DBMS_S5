CREATE OR REPLACE TRIGGER trigger3
BEFORE INSERT OR UPDATE ON "RETURN"
FOR EACH ROW
DECLARE 
    reserve_count NUMBER;
BEGIN
    -- Count the number of reserves for the ISBN in the new row
    SELECT COUNT(*) INTO reserve_count 
    FROM "RETURN"
    WHERE isbn = :NEW.ISBN;
    
    -- Check if the reserve count is greater than or equal to 2
    IF reserve_count >= 2 THEN
        RAISE_APPLICATION_ERROR(-20006, 'Cannot have more than 2 reserves for this ISBN');
    END IF;
END;
/

CREATE OR REPLACE TRIGGER reserveperday
BEFORE INSERT ON return
FOR EACH ROW
DECLARE
reservation_exists INT;
BEGIN
SELECT COUNT(*)
INTO reservation_exists
FROM return
WHERE USER_ID = :NEW.USER_ID
AND RESERVE_DATE = :NEW.RESERVE_DATE;
IF reservation_exists > 0 THEN
DBMS_OUTPUT.PUT_LINE('User ' || :NEW.USER_ID || ' already has a reservation on ' || :NEW.RESERVE_DATE);
END IF;
END;
/
